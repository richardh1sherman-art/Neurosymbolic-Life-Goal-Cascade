\section{Multi-Agent Weather-Adaptive Swarm Invariants (Tree \#12)}
To capture the mathematical intuition of distributed control, we model a swarm of $N=6$ vehicles navigating a continuous, turbulent space-time weather manifold.

\subsection{Agent Kinematics (ODE Layer)}
Each drone $i \in \{1, \dots, N\}$ is governed by decoupled 2D double-integrator dynamics:
\begin{equation}
\mathbf{\dot{p}}_i(t) = \mathbf{v}_i(t), \quad \mathbf{\dot{v}}_i(t) = \mathbf{u}_i(t)
\end{equation}
where $\mathbf{p}_i = [x_i, y_i]^T \in \mathbb{R}^2$ is the position vector, $\mathbf{v}_i \in \mathbb{R}^2$ is the velocity vector, and $\mathbf{u}_i \in \mathbb{R}^2$ is the acceleration control command vector.

\subsection{Turbulent Field Input (PDE Layer)}
The localized weather disturbance is governed by a 2D hyperbolic wave equation tracking a scalar intensity field $\phi(x, y, t)$ deformed by a non-linear Navier-Stokes advection velocity layer $\mathbf{v}_{\text{turb}}$ satisfying a Kolmogorov power law:
\begin{equation}
\frac{\partial^2 \phi}{\partial t^2} + \mathbf{v}_{\text{turb}} \cdot \nabla \left(\frac{\partial \phi}{\partial t}\right) = c^2 \nabla^2 \phi
\end{equation}

\subsection{The Multi-Objective Control Law}
The total acceleration command vector $\mathbf{u}_i(t)$ resolves three distinct geometric and physical constraints:
\begin{equation}
\mathbf{u}_i = \mathbf{u}_i^{\text{formation}} + \mathbf{u}_i^{\text{weather}} + \mathbf{u}_i^{\text{collision}}
\end{equation}

\subsubsection{Term 1: Consensus Formation Keeping}
Maintains the rigid geometric formation specified by structural neighbor offsets $\boldsymbol{\delta}_{ij}$:
\begin{equation}
\mathbf{u}_i^{\text{formation}} = -k_p \sum_{j=1}^{N} a_{ij} \left( \mathbf{p}_i - \mathbf{p}_j - \boldsymbol{\delta}_{ij} \right) - k_v \sum_{j=1}^{N} a_{ij} \left( \mathbf{v}_i - \mathbf{v}_j \right)
\end{equation}
where $k_p, k_v > 0$ are fixed feedback gains, and $a_{ij} = 1$ models the static all-to-all perfect communication link.

\subsubsection{Term 2: Wavelet Filtered Feedforward}
Extracts localized, scale-invariant spatial gradients using a Discrete Wavelet Transform (DWT) operator to prune out high-frequency chattering noise:
\begin{equation}
\mathbf{u}_i^{\text{weather}} = -k_w \cdot \text{DWT}\left(\nabla\phi(\mathbf{p}_i, t)\right)
\end{equation}

\subsubsection{Term 3: Artificial Potential Field Collision Avoidance}
Guarantees safety constraints by enforcing an unresolvable repulsion boundary if the inter-agent Euclidean distance $d_{ij} = \|\mathbf{p}_i - \mathbf{p}_j\|$ violates a critical threshold:
\begin{equation}
\mathbf{u}_i^{\text{collision}} = - \sum_{j \neq i} \nabla_{\mathbf{p}_i} V_{ij}(d_{ij})
\end{equation}
where the potential barrier function $V_{ij}(d_{ij})$ is defined over the influence range $d_{\text{influence}}$ and critical crash radius $r_{\text{safe}}$ as:
\begin{equation}
V_{ij}(d_{ij}) = 
\begin{cases} 
\frac{1}{2} \eta \left( \frac{1}{d_{ij} - r_{\text{safe}}} - \frac{1}{d_{\text{influence}} - r_{\text{safe}}} \right)^2 & \text{if } d_{ij} < d_{\text{influence}} \\ 
0 & \text{if } d_{ij} \ge d_{\text{influence}} 
\end{cases}
\end{equation}
with scaling parameter $\eta > 0$.
